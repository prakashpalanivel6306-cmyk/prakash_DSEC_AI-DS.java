5. Find the First Repeating Element

java
class Main {
    public static void main(String[] args) {
        int[] a = {4, 2, 7, 8, 2, 5, 4};
        int result = -1;

        for (int i = 0; i < a.length; i++) {
            for (int j = i + 1; j < a.length; j++) {
                if (a[i] == a[j]) {
                    result = a[i];
                    break;
                }
            }

            if (result != -1) {
                break;
            }
        }

        System.out.println("First Repeating Element = " + result);
    }
}



6. Find the First Non-Repeating Element

java
class Main {
    public static void main(String[] args) {
        int[] a = {2, 3, 4, 2, 3, 6, 7};
        int result = -1;

        for (int i = 0; i < a.length; i++) {
            int count = 0;

            for (int j = 0; j < a.length; j++) {
                if (a[i] == a[j]) {
                    count++;
                }
            }

            if (count == 1) {
                result = a[i];
                break;
            }
        }

        System.out.println("First Non-Repeating Element = " + result);
    }
}

8. Reverse Array Without Using Another Array

java
import java.util.Arrays;

class Main {
    public static void main(String[] args) {
        int[] a = {1, 2, 3, 4, 5};
        int left = 0;
        int right = a.length - 1;

        while (left < right) {
            int temp = a[left];
            a[left] = a[right];
            a[right] = temp;

            left++;
            right--;
        }

11. Find Equilibrium Index

class Main {
    public static void main(String[] args) {
        int[] a = {1, 3, 5, 2, 2};
        int total = 0;
        int leftSum = 0;
        int index = -1;

        for (int x : a) {
            total += x;
        }

        for (int i = 0; i < a.length; i++) {
            total -= a[i];

            if (leftSum == total) {
                index = i;
                break;
            }

            leftSum += a[i];
        }

        System.out.println("Equilibrium Index = " + index);
    }
}

13. Find Majority Element

class Main {
    public static void main(String[] args) {
        int[] a = {2, 2, 1, 2, 3, 2, 2};
        int candidate = 0;
        int count = 0;

        for (int x : a) {
            if (count == 0) {
                candidate = x;
            }

            if (x == candidate) {
                count++;
            } else {
                count--;
            }
        }

        count = 0;

        for (int x : a) {
            if (x == candidate) {
                count++;
            }
        }

        if (count > a.length / 2) {
            System.out.println("Majority Element = " + candidate);
        } else {
            System.out.println("No Majority Element");
        }
    }

17. Find the Smallest Positive Missing Number

class Main {
    public static void main(String[] args) {
        int[] a = {3, 4, -1, 1};
        int n = a.length;

        for (int i = 0; i < n; i++) {
            while (a[i] >= 1 && a[i] <= n
                    && a[a[i] - 1] != a[i]) {
                int temp = a[i];
                a[i] = a[temp - 1];
                a[temp - 1] = temp;
            }
        }

        int missing = n + 1;

        for (int i = 0; i < n; i++) {
            if (a[i] != i + 1) {
                missing = i + 1;
                break;
            }
        }

        System.out.println(missing);
    }
}
