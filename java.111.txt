Question 8: Age and City

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int age = sc.nextInt();
        String city = sc.next();

        System.out.println("Age: " + age);
        System.out.println("City: " + city);

        sc.close();
    }
}


Question 9: Student Registration Form
Use next() for single-word inputs and nextLine() for the student's full name.

java
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        String name = sc.nextLine();
        int rollNumber = sc.nextInt();
        String department = sc.next();
        double percentage = sc.nextDouble();

        System.out.println("STUDENT REGISTRATION");
        System.out.println("Name: " + name);
        System.out.println("Roll Number: " + rollNumber);
        System.out.println("Department: " + department);
        System.out.println("Percentage: " + percentage);

        sc.close();
    }
}

Question 11: Online Shopping Address Form
The customer's full name and street address contain spaces, so nextLine() is used to read them.

java
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        String name = sc.nextLine();
        String address = sc.nextLine();
        String city = sc.nextLine();
        int pinCode = sc.nextInt();
        boolean deliveryPreference = sc.nextBoolean();

        System.out.println("DELIVERY DETAILS");
        System.out.println("Customer Name: " + name);
        System.out.println("Street Address: " + address);
        System.out.println("City: " + city);
        System.out.println("PIN Code: " + pinCode);
        System.out.println("Delivery Preference: " + deliveryPreference);

        sc.close();
    }
}

10. Discount Calculator

import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        double amount = sc.nextDouble();

        double finalAmount = amount >= 2000 ? amount * 0.90 : amount;
        System.out.printf("%.2f%n", finalAmount);
    }
